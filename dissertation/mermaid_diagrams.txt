erDiagram
    USERS {
        int user_id PK
        varchar username
        varchar email UK
        varchar password_hash
        boolean is_verified
        varchar profile_picture_url
        timestamp created_at
        timestamp updated_at
    }

    SUBSCRIPTIONS {
        int subscription_id PK
        int user_id FK
        enum tier "free | pro | business"
        int daily_quota
        int max_input_px
        timestamp started_at
        timestamp expires_at
    }

    PROCESSING_LOGS {
        int log_id PK
        int user_id FK
        varchar model_type
        int scale_factor
        int input_width
        int input_height
        int output_width
        int output_height
        int processing_time_ms
        varchar cloudinary_url
        timestamp created_at
    }

    SESSIONS {
        int session_id PK
        int user_id FK
        varchar jwt_token_hash
        varchar ip_address
        varchar user_agent
        timestamp created_at
        timestamp expires_at
    }
